  add_link_options(-fsanitize=thread)

function(add_worker target source_dir)
  set(worker_root "${CMAKE_SOURCE_DIR}/workers/${source_dir}")
  file(GLOB worker_sources CONFIGURE_DEPENDS "${worker_root}/src/*.cpp")
  list(FILTER worker_sources EXCLUDE REGEX "/main\\.cpp$|/worker_runtime\\.cpp$|/thread_pool\\.cpp$|/retry_coordinator\\.cpp$|/redis_stream\\.cpp$")
  add_library(${target}_core ${worker_sources})
  target_include_directories(${target}_core PUBLIC "${worker_root}/src" ${PQXX_INCLUDE_DIRS} ${HIREDIS_INCLUDE_DIRS})
  target_link_libraries(${target}_core PUBLIC task_lifecycle Boost::system CURL::libcurl nlohmann_json::nlohmann_json ${PQXX_LIBRARIES} ${HIREDIS_LIBRARIES})
  target_compile_options(${target}_core PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(${target} "${worker_root}/src/main.cpp")
  target_link_libraries(${target} PRIVATE ${target}_core)
  if(BUILD_TESTING)
    file(GLOB worker_tests CONFIGURE_DEPENDS "${worker_root}/tests/unit/*.cpp")
    if(worker_tests)
      add_executable(${target}_tests ${worker_tests})
      target_link_libraries(${target}_tests PRIVATE ${target}_core GTest::gtest_main)
      target_include_directories(${target}_tests PRIVATE "${worker_root}/src")
      include(GoogleTest)
      gtest_discover_tests(${target}_tests DISCOVERY_MODE PRE_TEST)
    endif()
    file(GLOB contract_tests CONFIGURE_DEPENDS "${worker_root}/tests/contract/*.cpp")
    if(contract_tests)
      add_executable(${source_dir}_repository_contract_tests ${contract_tests})
      target_link_libraries(${source_dir}_repository_contract_tests PRIVATE ${target}_core GTest::gtest_main)
      target_include_directories(${source_dir}_repository_contract_tests PRIVATE "${worker_root}/src")
      gtest_discover_tests(${source_dir}_repository_contract_tests DISCOVERY_MODE PRE_TEST)
    endif()
  endif()
endfunction()

add_worker(screening-worker screening)
add_worker(definitive-task-worker definitive_task)
