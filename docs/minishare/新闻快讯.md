import minishare as ms

# 调用示例 - 快讯
token = "授权码"
df = ms.pro_api(token).news(
    start_date='2025-12-29 00:00:00',
    end_date='2025-12-29 24:00:00',
    src='cls',  # 可选：sina/wallstreetcn/10jqka/cls/eastmoney
    limit=500,
    offset=0
)
print(df)
参数说明
参数	类型	说明
start_date	string	开始时间，格式：YYYY-MM-DD HH:mm:ss
end_date	string	结束时间，格式：YYYY-MM-DD HH:mm:ss
src	string	可选。新闻来源：sina-新浪财经，wallstreetcn-华尔街见闻，10jqka-同花顺，cls-财联社，eastmoney-东方财富
limit	int	可选。单次返回条数上限，默认 1500；最大 1500（超出视为 1500）
offset	int	可选。跳过前 offset 条，用于分页；与 limit 配合分批拉取
